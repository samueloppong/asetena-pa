-- Enable PostGIS for geospatial location queries
CREATE EXTENSION IF NOT EXISTS postgis;

-- 1. USERS & ROLES
CREATE TYPE user_role AS ENUM ('renter', 'landlord', 'admin');
CREATE TYPE verification_status AS ENUM ('unverified', 'pending', 'verified', 'rejected');

CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email VARCHAR(255) UNIQUE NOT NULL,
    phone_number VARCHAR(20) UNIQUE NOT NULL,
    full_name VARCHAR(100) NOT NULL,
    role user_role DEFAULT 'renter',
    ghana_card_number VARCHAR(20) UNIQUE,
    id_verification_status verification_status DEFAULT 'unverified',
    avatar_url TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 2. PROPERTIES
CREATE TYPE property_type AS ENUM ('apartment', 'house', 'chamber_and_hall', 'single_room', 'short_stay');
CREATE TYPE payment_frequency AS ENUM ('monthly', 'six_months', 'one_year', 'two_years');

CREATE TABLE properties (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    landlord_id UUID REFERENCES users(id) ON DELETE CASCADE,
    title VARCHAR(150) NOT NULL,
    description TEXT NOT NULL,
    property_type property_type NOT NULL,
    
    -- Pricing
    price_per_month NUMERIC(10, 2) NOT NULL, -- GHS Currency
    advance_period payment_frequency NOT NULL,
    security_deposit NUMERIC(10, 2) DEFAULT 0.00,
    
    -- Location
    region VARCHAR(50) NOT NULL, -- e.g., Greater Accra, Ashanti
    city_town VARCHAR(50) NOT NULL, -- e.g., Accra, Kumasi
    neighborhood VARCHAR(50) NOT NULL, -- e.g., East Legon, Cantonments, Osu
    digital_address_ghanapost VARCHAR(15), -- e.g., GA-183-9020
    coordinates GEOGRAPHY(POINT, 4326),
    
    -- Property Specs
    bedrooms INT NOT NULL,
    bathrooms INT NOT NULL,
    square_footage INT,
    
    -- Verification & Status
    is_direct_owner_verified BOOLEAN DEFAULT FALSE,
    land_title_document_url TEXT,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 3. PROPERTY AMENITIES
CREATE TABLE property_amenities (
    property_id UUID REFERENCES properties(id) ON DELETE CASCADE,
    has_polytank_water BOOLEAN DEFAULT FALSE,
    has_standby_generator BOOLEAN DEFAULT FALSE,
    has_air_conditioning BOOLEAN DEFAULT FALSE,
    is_walled_and_gated BOOLEAN DEFAULT FALSE,
    has_self_meter BOOLEAN DEFAULT TRUE, -- Separate ECG meter
    has_parking_space BOOLEAN DEFAULT FALSE,
    PRIMARY KEY (property_id)
);

-- 4. PROPERTY MEDIA & VIEWS (Rooms & Categories)
CREATE TYPE room_category AS ENUM ('living_room', 'bedroom', 'kitchen', 'bathroom', 'balcony', 'exterior', 'street_view');

CREATE TABLE property_media (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    property_id UUID REFERENCES properties(id) ON DELETE CASCADE,
    media_url TEXT NOT NULL,
    media_type VARCHAR(10) CHECK (media_type IN ('image', 'video')),
    category room_category NOT NULL,
    display_order INT DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 5. TOUR SCHEDULES & IN-APP BOOKINGS
CREATE TYPE tour_status AS ENUM ('requested', 'confirmed', 'completed', 'cancelled');

CREATE TABLE physical_tours (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    property_id UUID REFERENCES properties(id) ON DELETE CASCADE,

🛡️ Business Logic & Security Rules
Anti-Scam Guardrail: Landlords cannot receive direct payments in-app until their identity (Ghana Card) is verified by the admin system or automated API check.

Zero Agent Fees Policy: Listings uploaded by third-party agents without direct property ownership or agency authorization are flagged and removed.

ECG & Water Transparency: Every property listing must explicitly state the utility situation (e.g., Self ECG Prepaid Meter vs Shared Meter, and Polytank presence).

How do we verify home ownership of people listing their properties on this app?

To ensure that listed properties on ShelterGH are strictly direct-from-owner and free of middleman scams or fraudulent postings, the app implements a 4-Tier Tiered Property Ownership Verification Protocol.

This framework combines real-time digital identity checks (Ghana Card), document verification, physical on-ground geotagging, and optional official government registry checks.

🛡️ 4-Tier Homeowner Verification Framework
Plaintext
[ Step 1: Identity KYC ] ──► [ Step 2: Proof of Title/Lease ] ──► [ Step 3: Geo-Video Check ] ──► [ Step 4: Official Registry (Opt.) ]
    Ghana Card + Facial            Land Indenture / Title /            On-Site Biometric &                 Lands Commission Search
    Real-Time Liveness             Utility Bill Matching Name          Live GPS Tagged Video Tour          (For Premium Direct Badge)
Tier 1: Identity & KYC Verification (Ghana Card API)
Before a landlord can post a property, their personal identity must be legally established.

Ghana Card Verification API: The app integrates with the NIA (National Identification Authority) via identity verification APIs (such as Smile Identity, YouVerify, or Dojah).

Liveness & Facial Match: The landlord takes a live selfie in-app. The system performs biometric face-matching against their Ghana Card record.

Result: Ensures the person creating the listing is an actual verified individual residing in Ghana, preventing anonymous or fake profiles.

Tier 2: Document Proof of Ownership / Authorization
The landlord uploads proof of authority to rent out the specific house or unit.

For Property Owners:
Land Title Register / Site Plan / Indenture: Uploaded in-app. The primary name on the deed must match the verified Ghana Card name from Tier 1.

Utility Bill Match (ECG / Ghana Water): Landlord uploads a recent ECG Prepaid meter receipt or Ghana Water bill showing the property’s physical address matched with their name.

For Family-Owned Properties / Authorized Representatives:
Power of Attorney / Family Consent Form: If an adult child or family administrator is listing the home on behalf of an elderly relative, they must upload a signed consent form + the primary owner's Ghana Card.

Tier 3: Geotagged "In-Person" Live Video Inspection
To eliminate fake agents who upload random downloaded photos from the internet, the app requires proof of physical possession.

In-App Live Video Recording: The landlord must record a short 30-second walkthrough video directly inside the app (gallery uploads disabled for this step).

GPS & Digital Address Verification:

The app extracts EXIF/GPS coordinates during recording and checks it against the GhanaPost GPS Digital Address (e.g., GA-183-9020).

If the listing claims to be in East Legon, but the upload location pin resolves to Kasoa, the system flags it for review.

Tier 4: Lands Commission Official Search Integration (For "Gold Verified" Badge)
For maximum trust on high-value properties (e.g., Cantonments, Airport Residential, East Legon), landlords can request an Official Lands Commission Title Search.

The app integrates with the Lands Commission Online Portal (onlineservices.lc.gov.gh).

The system submits the parcel/job number to verify that the property title is officially registered, unencumbered, and not tied up in litigation.

Badge Granted: 🟢 "Gold Verified Owner" display badge on the listing.
